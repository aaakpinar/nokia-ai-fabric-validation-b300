  neighbor OVERLAY soft-reconfiguration inbound

!

 match extcommunity OPENSTACK-VNI
exit
!

 match extcommunity OPENSTACK-VNI
exit
!

 match ip address prefix-list ACCEPT-DEFAULT
exit
!

exit
!
